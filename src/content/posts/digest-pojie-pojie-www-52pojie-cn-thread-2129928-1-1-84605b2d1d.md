int register_item(int idx)
{
    if (idx = item_count) return 0;
    if (items[idx].vk == 0) return 0;

    int id = HOTKEY_BASE + items[idx].uid;
    if (RegisterHotKey(win, id, items[idx].mods, items[idx].vk)) {
        items[idx].hk_id = id;
        return 1;
    }
    return 0;
}
